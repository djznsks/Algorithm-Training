/*
基本计算器

链接：
https://leetcode.cn/problems/basic-calculator/

类型：
递归、表达式解析、栈思想

解题思路：

使用递归函数 parseExpr 解析表达式。

核心思想：

parseExpr  // 处理加减（最低优先级）
parseTerm  // 处理乘除（中等优先级）  
getNumber  // 处理数字/括号（最高优先级）

时间复杂度：
O(n)

空间复杂度：
O(n)
（递归调用栈）


注意：
1. 括号需要递归处理
2. 一元负号：
   例如：
   -(2+3)
   -5

3. parseExpr 返回当前括号结果

*/

```cpp
class Solution {
public:
    int calculate(string s) {
        int i = 0;
        return parseExpr(s, i);
    }


private:

    // 处理 + -
    int parseExpr(const string& s, int& i) {

        int result = parseTerm(s, i);

        while (i < s.length()) {

            while (i < s.length() && s[i] == ' ')
                i++;


            if (s[i] == '+') {

                i++;
                result += parseTerm(s, i);

            }
            else if (s[i] == '-') {

                i++;
                result -= parseTerm(s, i);

            }
            else {

                break;

            }

        }

        return result;
    }



    // 处理 * /
    int parseTerm(const string& s, int& i) {

        int result = getNumber(s, i);


        while (i < s.length()) {


            while (i < s.length() && s[i] == ' ')
                i++;


            if (s[i] == '*') {

                i++;
                result *= getNumber(s, i);

            }
            else if (s[i] == '/') {

                i++;
                result /= getNumber(s, i);

            }
            else {

                break;

            }

        }


        return result;

    }



    // 获取一个数字
    // 支持：
    // 1. 普通数字
    // 2. 一元正负号
    // 3. 括号表达式

    int getNumber(const string& s, int& i) {


        while (i < s.length() && s[i] == ' ')
            i++;



        int sign = 1;


        // 处理一元符号
        if (i < s.length() &&
            (s[i] == '+' || s[i] == '-')) {


            if (s[i] == '-')
                sign = -1;


            i++;


            while (i < s.length() && s[i] == ' ')
                i++;

        }



        // 括号
        if (i < s.length() && s[i] == '(') {

            i++;

            int result = parseExpr(s, i);


            // 跳过 ')'
            if (i < s.length() && s[i] == ')')
                i++;


            return sign * result;

        }



        // 普通数字
        int num = 0;


        while (i < s.length() && isdigit(s[i])) {

            num = num * 10 + (s[i] - '0');

            i++;

        }


        return sign * num;

    }

};
