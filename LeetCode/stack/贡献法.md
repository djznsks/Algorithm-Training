给定一个整数数组 arr，找到 min(b) 的总和，其中 b 的范围为 arr 的每个（连续）子数组。

由于答案可能很大，因此 返回答案模 10^9 + 7 。


算法思想：贡献法
核心：通过个体分析他所在的所有集体

```cpp
class Solution {
public:
    int sumSubarrayMins(vector<int>& arr) {
        const int MOD = 1e9 + 7;
        int n = arr.size();
        long long ans = 0;
        stack<int> st;  // 存储索引，栈内元素值单调递增
        
        // 遍历数组，i = n 时用 0 作为哨兵
        for (int i = 0; i <= n; i++) {
            // 当前值：如果是最后一个位置，设为 0（比所有元素都小）
            int curVal = (i == n) ? 0 : arr[i];
            
            // 当栈不为空，且栈顶元素 >= 当前值
            // 说明栈顶元素的右边界就是 i
            while (!st.empty() && arr[st.top()] > curVal) {
                // 要结算的元素
                int mid = st.top();
                st.pop();
                
                // 左边界：栈顶元素（弹出后的栈顶）
                // 如果栈为空，左边界为 -1
                int left = st.empty() ? -1 : st.top();
                
                // 右边界：当前索引 i
                int right = i;
                
                // 计算贡献
                long long leftCount = mid - left;    // 左边可选起点数
                long long rightCount = right - mid;  // 右边可选终点数
                ans = (ans + arr[mid] * leftCount % MOD * rightCount) % MOD;
            }
            
            // 将当前索引入栈（除了哨兵）
            if (i < n) st.push(i);
        }
        
        return (int)ans;
    }
};
