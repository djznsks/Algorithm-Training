class PrefixXOR {
private:
    vector<int> pref;
public:
    // 构建前缀异或数组(arr为数组)
    PrefixXOR(vector<int>& arr) {
        int n = arr.size();
        pref.resize(n + 1, 0);
        for (int i = 0; i < n; i++) {
            pref[i + 1] = pref[i] ^ arr[i];
        }
    }
    // 构建前缀异或数组(arr为字符串)
    PrefixXOR(vector<int>& arr) {
        int n = arr.size();
        pref.resize(n + 1, 0);
        for (int i = 0; i < n; i++) {
            pref[i + 1] = pref[i] ^ (1<<(arr[i]-'a'));
        }
    }
    // 查询区间 [l, r] 的异或值（闭区间）
    int query(int l, int r) {
        return pref[r + 1] ^ pref[l];
    }
};
