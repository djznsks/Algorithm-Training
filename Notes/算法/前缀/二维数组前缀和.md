用途：求二维数组子矩阵之和；

```cpp
//sum[i][j]代表左上角为arr[0][0],右下角为arr[i-1][j-1]的矩阵元素和

for(int i = 1; i <= m; i++) {
        for(int j = 1; j <= n; j++) {
            cin >> arr[i][j];
            sum[i][j] = sum[i][j-1] + sum[i-1][j] + arr[i][j] - sum[i-1][j-1];
        }
    }
    
    cin >> q;
    while(q--) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        // 查询子矩阵和
        cout << sum[x2+1][y2+1] - sum[x1][y2+1] - sum[x2+1][y1] + sum[x1][y1] << "\n";
    }
    return 0;
}
