
例一:
给你一个整数数组 `nums` 和一个整数 `k` ，请你统计并返回 _该数组中和为 `k` 的子数组的个数_ 。
(子数组是数组中元素的连续非空序列)

```cpp

class Solution {

public:

    int subarraySum(vector<int>& nums, int k) {

        int n=nums.size();

        vector<int>pre(n+1,0);

        unordered_map<int,int>hash;

        hash[0]++;

        int ans=0;

        for(int i=1;i<=n;i++)

        {

            pre[i]=pre[i-1]+nums[i-1];

            int target=pre[i]-k;

            if(hash.find(target)!=hash.end()) ans+=hash[target];

            hash[pre[i]]++;

        }

        return ans;

    }

};
```

例二：
给定一个二进制数组 `nums` , 找到和为0的最长连续子数组，并返回该子数组的长度。
```cpp

 int findMaxLength(vector<int>& nums) {

        int n=nums.size();

        change(nums,n);

        unordered_map<int,int>hash;

        int ans=0;

        vector<int>pre(n+1,0);

        hash[0]=0;

        for(int i=1;i<=n;i++)

        {

            pre[i]=pre[i-1]+nums[i-1];

            if(hash.find(pre[i])!=hash.end()) ans=max(ans,i-hash[pre[i]]);

            else hash[pre[i]]=i;

        }

        return ans;

    }
