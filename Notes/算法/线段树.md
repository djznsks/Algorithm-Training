
线段树是一种**高效处理区间查询、区间修改**的数据结构，支持：

1. 单点修改 / 区间修改
2. 区间求和 / 区间最大值 / 区间最小值查询
3. 时间复杂度：**O(log n)**（修改 / 查询）

下面给你**最常用的区间和 + 懒标记（延迟更新）** 完整版代码，可直接使用。

```cpp

#include <iostream>
#include <vector>
using namespace std;

// 线段树类封装
class SegmentTree {
private:
    vector<long long> tree;  // 线段树数组
    vector<long long> lazy;  // 懒标记数组（区间修改必备）
    int n;                  // 原数组长度

    // 递归构建线段树
    // node: 当前节点编号  l,r: 当前节点覆盖区间
    void build(int node, int l, int r, vector<long long>& arr) {
        if (l == r) {
            // 叶子节点，直接赋值
            tree[node] = arr[l];
            return;
        }
        int mid = (l + r) / 2;
        build(2*node, l, mid, arr);      // 左子树
        build(2*node+1, mid+1, r, arr);  // 右子树
        // 回溯更新父节点 = 左孩子 + 右孩子
        tree[node] = tree[2*node] + tree[2*node+1];
    }

    // 懒标记下传（把当前节点的修改传递给子节点）
    void pushDown(int node, int l, int r) {
        if (lazy[node] == 0) return;  // 无标记，直接返回
        
        int mid = (l + r) / 2;
        int left = 2*node;
        int rightNode = 2*node+1;

        // 更新左孩子
        tree[left] += lazy[node] * (mid - l + 1);
        lazy[left] += lazy[node];
        
        // 更新右孩子
        tree[rightNode] += lazy[node] * (r - mid);
        lazy[rightNode] += lazy[node];
        
        // 清空当前节点懒标记
        lazy[node] = 0;
    }

    // 区间修改：[ul, ur] 区间全部加 val
    void rangeAdd(int node, int l, int r, int ul, int ur, long long val) {
        // 当前区间完全在修改区间内
        if (ul <= l && r <= ur) {
            tree[node] += val * (r - l + 1);
            lazy[node] += val;
            return;
        }
        // 下传懒标记
        pushDown(node, l, r);
        int mid = (l + r) / 2;
        if (ul <= mid) rangeAdd(2*node, l, mid, ul, ur, val);
        if (ur > mid) rangeAdd(2*node+1, mid+1, r, ul, ur, val);
        // 回溯更新
        tree[node] = tree[2*node] + tree[2*node+1];
    }

    // 区间查询：查询 [ql, qr] 区间和
    long long rangeQuery(int node, int l, int r, int ql, int qr) {
        // 当前区间完全在查询区间内
        if (ql <= l && r <= qr) {
            return tree[node];
        }
        pushDown(node, l, r);
        int mid = (l + r) / 2;
        long long res = 0;
        if (ql <= mid) res += rangeQuery(2*node, l, mid, ql, qr);
        if (qr > mid) res += rangeQuery(2*node+1, mid+1, r, ql, qr);
        return res;
    }

public:
    // 构造函数：传入原数组初始化
    SegmentTree(vector<long long>& arr) {
        n = arr.size() - 1;  // 约定数组从 1 开始
        tree.resize(4 * n); // 线段树开 4 倍空间足够
        lazy.resize(4 * n, 0);
        build(1, 1, n, arr);
    }

    // 对外接口：区间加
    void add(int l, int r, long long val) {
        rangeAdd(1, 1, n, l, r, val);
    }

    // 对外接口：区间查询和
    long long query(int l, int r) {
        return rangeQuery(1, 1, n, l, r);
    }
};

// 测试主函数
int main() {
    // 测试数组：下标从 1 开始（方便线段树处理）
    vector<long long> arr = {0, 1, 2, 3, 4, 5}; 
    SegmentTree st(arr);

    // 查询 [1,3] 和：1+2+3=6
    cout << "区间 [1,3] 和：" << st.query(1, 3) << endl;

    // 区间 [2,4] 全部 +2
    st.add(2, 4, 2);
    cout << "区间 [2,4] +2 后" << endl;

    // 再次查询 [1,3]：1 + (2+2) + (3+2) = 10
    cout << "区间 [1,3] 和：" << st.query(1, 3) << endl;

    // 查询 [1,5] 和
    cout << "区间 [1,5] 和：" << st.query(1, 5) << endl;

    return 0;
}
