class Difference2D {
private:
    vector<vector<int>> diff;
    int n, m;
    
public:
    // 构造函数：原矩阵大小为 n x m，索引从 0 开始
    Difference2D(int n, int m) : n(n), m(m) {
        // 只需要 n+1 x m+1，索引范围 0..n, 0..m
        diff.assign(n + 1, vector<int>(m + 1, 0));
    }
    
    // 对子矩阵 (x1,y1) 到 (x2,y2) 加上 val（包含边界）
    // 所有坐标从 0 开始
    void add(int x1, int y1, int x2, int y2, int val) {
        diff[x1][y1] += val;
        diff[x1][y2 + 1] -= val;
        diff[x2 + 1][y1] -= val;
        diff[x2 + 1][y2 + 1] += val;
    }
    
    // 还原原矩阵，返回 n x m 大小
    vector<vector<int>> getResult() {
        vector<vector<int>> res(n, vector<int>(m, 0));
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                res[i][j] = diff[i][j];
                if (i > 0) res[i][j] += res[i-1][j];
                if (j > 0) res[i][j] += res[i][j-1];
                if (i > 0 && j > 0) res[i][j] -= res[i-1][j-1];
                
                //因为它的合成本质是前缀和，所以这个公式就是二位前缀和构造公式很类似，方便记忆
                
            }
        }
        return res;
    }
};
