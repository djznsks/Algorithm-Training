class Solution {

public:

    int minimumSum(vector<int>& nums) {

        int n=nums.size();

        vector<int>preleftmin(n),prerightmin(n);

        preleftmin[0]=nums[0],prerightmin[n-1]=nums[n-1];

        for(int i=1;i<n;i++) preleftmin[i]=min(preleftmin[i-1],nums[i]);

        for(int j=n-2;j>=0;j--) prerightmin[j]=min(prerightmin[j+1],nums[j]);
    }

};
