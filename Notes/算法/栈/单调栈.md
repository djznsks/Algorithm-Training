> [!info] 单调栈核心概念
> **单调栈**：一种特殊的栈结构，栈内元素保持单调递增或递减的顺序
> 
> **主要用途**：
> - 寻找下一个更大/更小元素
> - 解决区间最值问题
> - 计算柱状图中最大矩形
> - 去除重复字符保持字典序最小
> 
> **时间复杂度**：O(n)，每个元素最多入栈出栈一次

## 📚 基础模板

### 1. 单调递减栈（找下一个更大元素）
```cpp title="单调递减栈模板 - 找下一个更大元素"
vector<int> nextGreaterElement(vector<int>& nums) {
    int n = nums.size();
    vector<int> result(n, -1);  // 默认-1表示没有更大元素
    stack<int> stk;             // 单调递减栈（栈底到栈顶递减）
    
    for (int i = 0; i < n; i++) {
        // 当栈不为空且当前元素大于栈顶元素时
        while (!stk.empty() && nums[stk.top()] < nums[i]) {
            int idx = stk.top();  // 栈顶元素的索引
            stk.pop();
            result[idx] = nums[i];  // 当前元素是栈顶元素的下一个更大元素
        }
        stk.push(i);  // 将当前索引入栈
    }
    
    return result;
}
```

## 🎯 经典应用

### 1. LeetCode 496 - 下一个更大元素 I
```cpp title="下一个更大元素 I"
class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        stack<int> stk;
        unordered_map<int, int> nextGreater;  // 存储每个元素的下一个更大元素
        
        // 对nums2使用单调栈
        for (int num : nums2) {
            while (!stk.empty() && stk.top() < num) {
                nextGreater[stk.top()] = num;
                stk.pop();
            }
            stk.push(num);
        }
        // 栈中剩余元素没有下一个更大元素
        
        // 构建结果
        vector<int> result;
        for (int num : nums1) {
            if (nextGreater.count(num)) {
                result.push_back(nextGreater[num]);
            } else {
                result.push_back(-1);
            }
        }
        return result;
    }
};
```

### 2. LeetCode 739 - 每日温度
```cpp title="每日温度"
class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        int n = temperatures.size();
        vector<int> result(n, 0);
        stack<int> stk;  // 存储索引的单调递减栈
        
        for (int i = 0; i < n; i++) {
            // 当遇到更高温度时
            while (!stk.empty() && temperatures[stk.top()] < temperatures[i]) {
                int idx = stk.top();
                stk.pop();
                result[idx] = i - idx;  // 计算等待天数
            }
            stk.push(i);
        }
        
        return result;
    }
};
```

### 3. LeetCode 84 - 柱状图中最大的矩形
```cpp title="柱状图中最大的矩形"
class Solution {
public:
    int largestRectangleArea(vector<int>& heights) {
        int n = heights.size();
        stack<int> stk;  // 单调递增栈
        int maxArea = 0;
        
        // 添加哨兵节点，简化代码
        heights.insert(heights.begin(), 0);   // 头部添加0
        heights.push_back(0);                 // 尾部添加0
        n += 2;
        
        for (int i = 0; i < n; i++) {
            // 当当前高度小于栈顶高度时
            while (!stk.empty() && heights[stk.top()] > heights[i]) {
                int height = heights[stk.top()];
                stk.pop();
                
                // 计算宽度
                int left = stk.empty() ? 0 : stk.top() + 1;
                int right = i - 1;
                int width = right - left + 1;
                
                maxArea = max(maxArea, height * width);
            }
            stk.push(i);
        }
        
        return maxArea;
    }
};
```

### 4. LeetCode 42 - 接雨水
```cpp title="接雨水 - 单调栈解法"
class Solution {
public:
    int trap(vector<int>& height) {
        int n = height.size();
        if (n < 3) return 0;
        
        stack<int> stk;  // 单调递减栈（存索引）
        int water = 0;
        
        for (int i = 0; i < n; i++) {
            while (!stk.empty() && height[stk.top()] < height[i]) {
                int bottomIdx = stk.top();  // 凹槽底部
                stk.pop();
                
                if (stk.empty()) break;  // 没有左边界
                
                int leftIdx = stk.top();  // 左边界
                int rightIdx = i;         // 右边界
                
                // 计算水的高度和宽度
                int h = min(height[leftIdx], height[rightIdx]) - height[bottomIdx];
                int w = rightIdx - leftIdx - 1;
                
                water += h * w;
            }
            stk.push(i);
        }
        
        return water;
    }
};
```
## 🔄 环形数组处理

### LeetCode 503 - 下一个更大元素 II
```cpp title="环形数组下一个更大元素"
class Solution {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        int n = nums.size();
        vector<int> result(n, -1);
        stack<int> stk;
        
        // 遍历两遍数组模拟环形
        for (int i = 0; i < 2 * n; i++) {
            int idx = i % n;  // 实际索引
            
            while (!stk.empty() && nums[stk.top()] < nums[idx]) {
                int topIdx = stk.top();
                stk.pop();
                result[topIdx] = nums[idx];
            }
            
            // 只在第一遍遍历时入栈
            if (i < n) {
                stk.push(idx);
            }
        }
        
        return result;
    }
};
```

## 🛠️ 通用模板

```cpp title="单调栈通用模板"
vector<int> monotonicStackTemplate(vector<int>& arr, bool increasing = false) {
    int n = arr.size();
    vector<int> result(n, -1);  // 根据需要初始化
    stack<int> stk;
    
    for (int i = 0; i < n; i++) {
        // 根据需求选择单调递增或递减
        if (increasing) {
            // 单调递增栈：找下一个更小元素
            while (!stk.empty() && arr[stk.top()] > arr[i]) {
                int idx = stk.top();
                stk.pop();
                result[idx] = arr[i];  // 或 i，根据需求
            }
        } else {
            // 单调递减栈：找下一个更大元素
            while (!stk.empty() && arr[stk.top()] < arr[i]) {
                int idx = stk.top();
                stk.pop();
                result[idx] = arr[i];  // 或 i，根据需求
            }
        }
        stk.push(i);
    }
    
    return result;
}
```
