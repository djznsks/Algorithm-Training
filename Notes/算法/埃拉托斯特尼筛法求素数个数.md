#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

// 埃氏筛统计素数个数
int count_primes_sieve(int n) {
    if (n < 2) return 0;
    // 动态分配数组（避免栈溢出，n最大1e7）
    bool *is_prime = (bool*)malloc((n + 1) * sizeof(bool));
    if (is_prime == NULL) {  // 内存分配失败处理
        printf("内存分配失败\n");
        exit(1);
    }
    // 初始化：所有数默认是素数
    for (int i = 0; i <= n; i++) {
        is_prime[i] = true;
    }
    is_prime[0] = is_prime[1] = false;  // 0和1非素数
    
    // 核心筛法逻辑
    for (int i = 2; i * i <= n; i++) {
        if (is_prime[i]) {  // 若i是素数，标记其所有倍数
            // 从i*i开始标记（更小的倍数已被之前的素数标记）
            for (int j = i * i; j <= n; j += i) {
                is_prime[j] = false;
            }
        }
    }
    
    // 统计素数个数
    int count = 0;
    for (int i = 2; i <= n; i++) {
        if (is_prime[i]) count++;
    }
    
    free(is_prime);  // 释放内存
    return count;
}

int main() {
    int n;
    scanf("%d", &n);
    int result = count_primes_sieve(n);
    printf("%d\n", result);
    return 0;
