
> [!code]- LeetCode 43: 字符串相乘（高精度乘法）
> **题目描述**: 给定两个以字符串形式表示的非负整数 `num1` 和 `num2`，返回它们的乘积，也用字符串表示。
> 
> **算法特点**:
> - 模拟手算乘法过程
> - 时间复杂度: O(m×n)
> - 空间复杂度: O(m+n)
> - 支持任意大整数相乘
> 
> **关键技巧**:
> 1. 使用数组暂存中间结果
> 2. 逆序遍历字符串（从个位开始计算）
> 3. 巧妙处理进位和前导零
> 
> ```cpp title="Solution.cpp - 字符串相乘"
> #include <vector>
> #include <string>
> using namespace std;
> 
> class Solution {
> public:
>     /**
>      * 字符串相乘：计算 num1 × num2
>      * @param num1 第一个数字字符串
>      * @param num2 第二个数字字符串
>      * @return 乘积的字符串表示
>      */
>     string multiply(string num1, string num2) {
>         // 1. 边界条件处理
>         if (num1 == "0" || num2 == "0") {
>             return "0";
>         }
>         
>         int m = num1.size(), n = num2.size();
>         // 结果最大长度为 m + n（如 99×99=9801，长度从2+2=4）
>         vector<int> result(m + n, 0);
> 
>         // 2. 模拟手算乘法
>         // 从个位开始计算（字符串逆序遍历）
>         for (int i = m - 1; i >= 0; i--) {
>             int digit1 = num1[i] - '0';  // 当前位数字
>             
>             for (int j = n - 1; j >= 0; j--) {
>                 int digit2 = num2[j] - '0';
>                 
>                 // 当前位置：result[i + j + 1] 是当前乘积要累加的位置
>                 // 为什么是 i+j+1？考虑索引：
>                 // num1[0]是最高位，num2[0]是最高位
>                 // 当i=m-1(个位), j=n-1(个位)时，i+j+1 = (m-1)+(n-1)+1 = m+n-1
>                 // 这是结果的个位位置（数组最后一个元素）
>                 int mul = digit1 * digit2;
>                 int sum = mul + result[i + j + 1];  // 加上之前的结果
> 
>                 result[i + j + 1] = sum % 10;  // 当前位置存个位
>                 result[i + j] += sum / 10;     // 进位加到前一位
>                 
>                 // 注意：这里可能会使 result[i+j] 暂时大于9
>                 // 但会在后续迭代中被处理（因为外层循环从右向左）
>             }
>         }
> 
>         // 3. 将数组转换为字符串，去除前导零
>         string res;
>         for (int num : result) {
>             // 跳过前导零：只有结果字符串为空且当前位为0时才跳过
>             if (!(res.empty() && num == 0)) {
>                 res.push_back(num + '0');  // 数字转字符
>             }
>         }
> 
>         // 4. 处理特殊情况：如果所有位都是0（如乘0的情况）
>         // 但前面已处理过0的情况，这里保留以防万一
>         return res.empty() ? "0" : res;
>     }
> };
> ```
