class FenwickTree {
private:
    vector<int> BIT;
    int n;
    
public:
    FenwickTree(int size) {
        n = size;
        BIT.resize(n + 1, 0);
    }
    
    int lowbit(int x) {
        return x & -x;  // 求x二进制中1的最低位
    }
    
    // 单点更新：在 idx 位置增加 delta
    void update(int idx, int delta) {
        while(idx <= n) {
            BIT[idx] += delta;
            idx += lowbit(idx);
        }
    }//原理：BIT[i]的正上方序列正好是BIT[i+lowbit(i)]
    
    // 前缀查询：查询 [1, idx] 的和（递归版）
    int query(int idx) {
        if(idx == 0) return 0;
        return query(idx - lowbit(idx)) + BIT[idx];
    }//原理：BIT[i]代表以i为结尾，长度为lowbit(i)序列的和
    
    // 区间查询：[l, r] 的和
    int rangeQuery(int l, int r) {
        return query(r) - query(l - 1);
    }
};

class Solution {

public:

    long long countMajoritySubarrays(vector<int>& nums, int target) {

        int n = nums.size();

        // 转换为 +1/-1 数组

        for(int i = 0; i < n; i++) {

            if(nums[i] == target) nums[i] = 1;

            else nums[i] = -1;

        }

        // 计算前缀和

        vector<int> pre(n + 1, 0);

        for(int i = 0; i < n; i++) {

            pre[i + 1] = pre[i] + nums[i];

        }

        // 离散化前缀和，用来当作字典查询pre数组中每个元素的大小排序

        vector<int> sorted_pre = pre;

        sort(sorted_pre.begin(), sorted_pre.end());

        sorted_pre.erase(unique(sorted_pre.begin(), sorted_pre.end()), sorted_pre.end());

        // Fenwick Tree

        int m = sorted_pre.size();

        vector<int> bit(m + 1, 0);

        auto update = [&](int idx, int val) {

            while(idx <= m) {

                bit[idx] += val;

                idx += idx & -idx;

            }

        };

        auto query = [&](int idx) {

           if(idx==0) return 0;
           return (query(idx-idx%(-idx))+BIT[idx]);

        };
		
		// BIT[i]存储pre中第i小（从小到大排序后第 i 个）元素的目前出现频次
		
        long long ans = 0;

        // 初始插入前缀和 0  pre[0]=0;

        int idx0 = lower_bound(sorted_pre.begin(), sorted_pre.end(), 0) - sorted_pre.begin() + 1;//+1是因为BIT树状数组是从1开始

        update(idx0, 1);

        for(int i = 1; i <= n; i++) {

            // 找到第一个 >= pre[i] 的位置（统计 < pre[i] 的个数）

            int idx = lower_bound(sorted_pre.begin(), sorted_pre.end(), pre[i]) - sorted_pre.begin();

            // idx 是索引，需要 +1 转成树状数组的位置

            ans += query(idx);  // query 统计 < pre[i] 的元素个数

            // 插入当前前缀和

            int pos = lower_bound(sorted_pre.begin(), sorted_pre.end(), pre[i]) - sorted_pre.begin() + 1;

            update(pos, 1);
            
            //pos等于idx+1，而且由于BIT是1开头，所以idx在这里代表的就不是lower_upper,而是小于PRE[I]中最大的

        }

        return ans;

    }

	};
