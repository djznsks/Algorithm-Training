矩阵快速幂求斐波那契：
 ```cpp
 
#include <iostream>
#include <cstring>
using namespace std;

typedef long long ll;
const int MOD = 1e9 + 7;

// 2x2 矩阵
struct Matrix {
    ll m[2][2];
    Matrix() {
        memset(m, 0, sizeof(m));
    }
};

// 矩阵乘法（带取模）
Matrix mul(Matrix a, Matrix b) {
    Matrix res;
    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            for (int k = 0; k < 2; k++)
                res.m[i][j] = (res.m[i][j] + a.m[i][k] * b.m[k][j]) % MOD;
    return res;
}

// 矩阵快速幂
Matrix matrix_pow(Matrix a, ll power) {
    // 单位矩阵（乘法单位元）
    Matrix res;
    res.m[0][0] = res.m[1][1] = 1;

    while (power > 0) {
        if (power & 1) res = mul(res, a);
        a = mul(a, a);
        power >>= 1;
    }
    return res;
}

// 求 F(n)，定义：F(0)=0, F(1)=1, F(2)=1, F(3)=2...
ll fib(ll n) {
    if (n == 0) return 0;   // F0
    if (n == 1) return 1;   // F1

    Matrix M;
    // 转移矩阵 [[1,1],[1,0]]
    M.m[0][0] = 1; M.m[0][1] = 1;
    M.m[1][0] = 1; M.m[1][1] = 0;

    // 核心：求 M^(n-1)
    Matrix ans = matrix_pow(M, n - 1);

    // 结果 = ans.m[0][0] * F1 + ans.m[0][1] * F0
    return (ans.m[0][0] * 1 + ans.m[0][1] * 0) % MOD;
}

int main() {
    ll n;
    cin >> n;
    cout << fib(n) << endl;
    return 0;
}
