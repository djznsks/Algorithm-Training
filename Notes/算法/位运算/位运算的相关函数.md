#include <bitset>
#include <iostream>
//统计二进制中1的个数的函数
int main() {
    unsigned int x = 29; // 11101
    // C++20 标准库
    std::cout << std::popcount(x) << std::endl; // 输出 4
    
    // 或使用 bitset
    std::cout << std::bitset<32>(x).count() << std::endl;
    
    // GCC/Clang 内置函数
    std::cout << __builtin_popcount(x) << std::endl;
}
