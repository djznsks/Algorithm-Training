查询矩阵第K小的数：

class Solution {

public:

    bool check(vector<vector<int>>& matrix,int mid,int n,int k)

    {

        int i=n-1;

        int j=0;

        int num=0;

        while(j<n&&i>=0)

        {

            if(matrix[i][j]<=mid)

            {

                num+=i+1;

                j++;

            }

            else i--;

        }

        return num>=k;

     }

    int kthSmallest(vector<vector<int>>& matrix, int k) {

        int n=matrix.size();

        int l=matrix[0][0];

        int r=matrix[n-1][n-1];

        while(l<r)

        {

            int mid = l + ((r - l) / 2);//细节

            cout<<mid<<" ";

            if(check(matrix,mid,n,k))

            {

                r=mid;

            }

            else l=mid+1;

        }

        return l;

    }

};
