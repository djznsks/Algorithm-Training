
# 环游加油站问题解法

## 代码实现
```cpp
#include <bits/stdc++.h>

using namespace std;
using ll = long long;
using ull = unsigned long long;
using db = double;
#define ff(n) fixed << setprecision(n)
#define PI acos(-1.0)

int main() {
    ios::sync_with_stdio(false), cin.tie(nullptr), cout.tie(nullptr);
    
    int n;
    cin >> n;
    
    // 三个数组分别存储：
    int number[n];  // 加油站可获得的油量
    int cost[n];    // 前往下一站的耗油量
    int res[n];     // 净收益（获得-消耗）
    
    // 输入数据
    for(int i = 0; i < n; i++) cin >> number[i];
    for(int i = 0; i < n; i++) cin >> cost[i];
    
    // 计算每个加油站的净收益
    for(int i = 0; i < n; i++) res[i] = number[i] - cost[i];
    
    // 使用滑动窗口/双指针法寻找可行起点
    int start = 0, end = 0, current = 0;
    
    // 遍历可能的起点，end可以绕回到数组开头
    while(start < n && end != start + n) {
        int cur = end % n;  // 取模实现环形数组
        current += res[cur];
        
        if(current < 0) {
            // 当前起点不可行，回退并尝试下一个起点
            current -= res[cur];  // 回退最后一步
            current -= res[start]; // 移除起点的影响
            start++;              // 尝试下一个起点
        } else {
            // 当前仍可行，继续前进
            end++;
        }
    }
    
    // 计算最终结果
    int ress = start > n ? -1 : start;  // 如果start超过n，说明无解
    cout << ress;
    
    return 0;
}
